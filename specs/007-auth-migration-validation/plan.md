```typescript
// JWT Bridge - converts Better Auth session to JWT
export async function GET(request: NextRequest) {
  const session = await auth.api.getSession({ headers: request.headers })

  if (!session?.user) {
    return NextResponse.json({ error: 'No active session' }, { status: 401 })
  }

  const token = jwt.sign(
    { sub: session.user.id, email: session.user.email },
    process.env.BETTER_AUTH_SECRET,
    { algorithm: 'HS256', expiresIn: '15m' }
  )

  return NextResponse.json({ token })
}
```

```python
@dataclass
class AuthenticatedUser:
    user_id: str
    email: str

async def get_current_user(
    authorization: Annotated[str | None, Header()] = None
) -> AuthenticatedUser:
    # 1. Check Authorization header
    if not authorization:
        raise HTTPException(status_code=401, detail="Authentication required")

    # 2. Validate Bearer format
    parts = authorization.split()
    if len(parts) != 2 or parts[0].lower() != "bearer":
        raise HTTPException(status_code=401, detail="Invalid authorization header format")

    # 3. Verify JWT signature and extract claims
    payload = jwt.decode(token, settings.jwt_secret, algorithms=[settings.jwt_algorithm])

    # 4. Return authenticated user
    return AuthenticatedUser(user_id=payload["sub"], email=payload["email"])
```

```typescript
class ApiClient {
  private async getJWT(): Promise<string> {
    const response = await fetch('/api/auth/jwt', {
      method: 'GET',
      credentials: 'include',  // Send Better Auth cookies
    })

    if (response.status === 401) {
      throw new Error('SESSION_INVALID')
    }

    const data = await response.json()
    return data.token
  }

  private async request<T>(url: string, options: RequestInit = {}): Promise<T> {
    const jwt = await this.getJWT()

    const response = await fetch(`${this.baseUrl}${url}`, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${jwt}`,
        ...options.headers,
      },
    })

    if (response.status === 401) {
      throw new Error('SESSION_INVALID')  // Trigger re-auth
    }

    return response.json()
  }
}
```
